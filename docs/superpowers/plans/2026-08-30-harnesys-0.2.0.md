# Harnesys 0.2.0 executable graph Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Deliver 0.2.0 executable graph per ROADMAP (`llm:generate → tool:call → end` on InMemory) reusing Harnyx tested modules for 06–07.

**Architecture:** Hybrid C — domain pure (`expr`, `validate/compile`), ports for models/tools/state, application graph (`llm`, `tool-call`, `assign/goto`, `run/start`), adapters for `ai` (private) and discover. `defineAgent` structural checks throw; `compile` pure plan; `check` probes runtime maps; `run` consumes `start` AsyncIterable<Event> with recorded commits.

**Tech Stack:** Bun + TypeScript 5.9 (strict, bundler, `types:["bun"]`), `ai` 7 + `@ai-sdk/*` (private), `ajv` 8 for ToolDefinition input, `biome` 2.5.6, No `T['field']`, file ≤300 lines.

**Spec:** `docs/superpowers/specs/2026-08-30-harnesys-0.2.0-design.md`

import Ajv from 'ajv';
import type { ToolDefinition } from '../ports/tools.ts';
const ajv = new Ajv({ strict:false, allErrors:true });
export function createToolRegistry(tools: ToolDefinition[] = []): Map<string,ToolDefinition> {
  const map = new Map<string,ToolDefinition>();
  for(const t of tools){ if(map.has(t.name)) throw new Error(`tool collision: ${t.name}`); map.set(t.name,t); }
  return map;
}
export function validateToolInput(schema:unknown, data:unknown): { ok:boolean; errors?:string } {
  const valid = ajv.validate(schema as never, data);
  if(valid) return { ok:true };
  return { ok:false, errors: ajv.errorsText(ajv.errors) };
}

import { evalWhen, evalExpr } from './expr-eval.ts';
import { runLlmGenerate } from './llm.ts';
import { executeToolCall } from './tool-call.ts';
export type GraphOpts = { agent:AgentDefinition; input:unknown; state:RuntimeState; permissions?:PermissionMap; paths?:PathsConfig; models:ProviderConfig[]|ModelsPort; toolRegistry:Map<string,ToolDefinition>; plan:ReturnType<typeof compile>['plan']; toolMessages:'barrier'|'ordered'; mergeState?:(k:string,a:unknown,b:unknown)=>unknown };
export async function* startGraph(opts:GraphOpts): AsyncIterable<Event> {
  // load snapshot = await state.load() ?? initial snapshot with sequence 0, state from state.initial evaluated under $input
  // runId=crypto.randomUUID(), steps=0, tokens=0, cursor={currentNodeId, phase}
  // loop: node = plan.nodes[currentNodeId]; handle per type:
  // core:start → state init, output={input}, advance via edge
  // core:end → output Expr eval → RunSuccess, commit recorded, yield done Event, break
  // llm:generate → llm binding via bindingOf, runLlmGenerate, state.messages append assistant, commit recorded (batch chunk if needed), output = llm result
  // tool:call → executeToolCall, $output.results, flush messages, commit recorded
  // control:assign → evaluate patch entries, apply replace per key via reducers map (omit=merge), output {patched}, commit recorded
